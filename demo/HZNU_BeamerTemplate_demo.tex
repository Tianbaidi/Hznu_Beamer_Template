\section{文本强调}
\begin{frame}{文本强调效果}
  \begin{block}{Beamer 强调命令}
    hznu.sty 定制了下列语义色彩的默认外观：
  \end{block}

  \vspace{.2cm}
  \begin{itemize}
    \item 这是普通正文文本 \texttt{\textbackslash normal text}（hznuInk）
    \item \alert{这是强调文本 \texttt{\textbackslash alert\{...\}}（hznuGold）}
    \item \structure{这是结构文本 \texttt{\textbackslash structure\{...\}}（hznuBlue）}
    \item 混合使用：在 \alert{警告} 中可以夹杂 \structure{结构} 信息
  \end{itemize}

  \vspace{.1cm}
  \begin{alertblock}{alertblock — 金色警告区块}
    使用 \texttt{\textbackslash alertblock} 环境展示警告或注意事项。
    hznuGold 用于标题底色，正文底色为 hznuGold!10!white。
  \end{alertblock}

  \begin{exampleblock}{exampleblock — 蓝色示例区块}
    使用 \texttt{\textbackslash exampleblock} 环境展示示例。
    hznuBlue 用于标题底色。
  \end{exampleblock}
\end{frame}

\section{数学公式}
\subsection{基本公式}
\begin{frame}{数学公式展示（一）—— 基本公式}
  \framesubtitle{行内公式与行间编号公式}
  
  \begin{block}{行内公式}
    行内公式嵌入文本流中，使用 \texttt{\$...\$ 或 \textbackslash(...\textbackslash)} 包裹：
    \begin{itemize}
      \item 质能方程：$E = mc^2$
      \item 三角恒等式：$\sin^2\theta + \cos^2\theta = 1$
      \item 欧拉公式：$e^{i\pi} + 1 = 0$
      \item 求和记号：$a_1 + a_2 + \cdots + a_n = \sum_{k=1}^{n} a_k$
    \end{itemize}
  \end{block}
\end{frame}
\begin{frame}{数学公式展示（二）—— 行间编号公式}
  \vspace{.2cm}
  \begin{block}{行间编号公式（\texttt{equation} 环境）}
    \begin{itemize}
      \item 拉格朗日量：
      \begin{equation}
        \mathcal{L} = -\frac{1}{4} F_{\mu\nu}F^{\mu\nu} + i\bar{\psi}\gamma^\mu D_\mu\psi
      \end{equation}
      \item 牛顿第二定律：
      \begin{equation}
        \mathbf{F} = m\mathbf{a} = m\frac{\mathrm{d}^2\mathbf{r}}{\mathrm{d}t^2}
      \end{equation}
    \end{itemize}
  \end{block}
\end{frame}
